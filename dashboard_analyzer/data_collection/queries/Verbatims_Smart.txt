DEFINE
    VAR __DS0FilterTable =
        TREATAS({"Business", "Economy", "Premium EC"}, 'Cabin_Master'[Cabin_Show])

    VAR __DS0FilterTable2 =
        TREATAS({"IB","YW"}, 'Company_Master'[Company])

    VAR __DS0FilterTable3 =
        TREATAS({"SH","LH"}, 'Haul_Master'[Haul_Aggr])

    VAR __DS0FilterTable4 =
        FILTER(
            KEEPFILTERS(VALUES('Date_Master'[Date])),
            AND(
                'Date_Master'[Date] >= DATE(__START_YEAR__, __START_MONTH__, __START_DAY__),
                'Date_Master'[Date] <= DATE(__END_YEAR__, __END_MONTH__, __END_DAY__)
            )
        )

    VAR __SmartVerbatims =
        TOPN(
            30,
            CALCULATETABLE(
                SELECTCOLUMNS(
                    surveys_maritz,
                    "Verbatim", surveys_maritz[nps_all_t],
                    "Route", surveys_maritz[route],
                    "Date", surveys_maritz[date_flight_local],
                    "NPS_Score", surveys_maritz[nps_100],
                    "NPS_Category", surveys_maritz[nps_category]
                ),
                __DS0FilterTable,
                __DS0FilterTable2,
                __DS0FilterTable3,
                __DS0FilterTable4,
                KEEPFILTERS(__NPS_CLASS_FILTER__)
            ),
            LEN([Verbatim]), DESC
        )

EVALUATE
    __SmartVerbatims
